```python
# preprocessing.py
from __future__ import annotations
from typing import Tuple
import numpy as np
from scipy.signal import butter, filtfilt, resample

from config import cfg
from data_loader import Dataset

def bandpass_filter(
    X: np.ndarray,
    fs: float,
    f_low: float,
    f_high: float,
    order: int = 5,
) -> np.ndarray:
    """
    Zero-phase bandpass (Butterworth + filtfilt)

    X: (n_trials, n_channels, n_samples)
    """
    nyq = 0.5 * fs
    b, a = butter(order, [f_low / nyq, f_high / nyq], btype="band")
    Xf = filtfilt(b, a, X, axis=-1)
    return Xf.astype(np.float32)

def resample_to_fs(dataset: Dataset, fs_new: float) -> Dataset:
    """
    將資料重採樣到 fs_new。
    TODO: 如果原本就 fs_new，可直接 return。
    """
    if abs(dataset.fs - fs_new) < 1e-5:
        return dataset

    n_trials, n_ch, n_samples = dataset.X.shape
    n_samples_new = int(dataset.X.shape[-1] * fs_new / dataset.fs)
    X_rs = resample(dataset.X, n_samples_new, axis=-1)
    return Dataset(
        X=X_rs.astype(np.float32),
        y=dataset.y,
        blocks=dataset.blocks,
        ch_names=dataset.ch_names,
        fs=fs_new,
    )

def crop_trials(dataset: Dataset, t_start: float, t_end: float) -> Dataset:
    """
    依時間範圍裁切 trial (相對於目前 X 的 0s)。

    TODO:
    - 根據你原始 epoch 的時間軸決定 t_start, t_end
    """
    fs = dataset.fs
    start_idx = int(round(t_start * fs))
    end_idx = int(round(t_end * fs))
    X_crop = dataset.X[..., start_idx:end_idx]
    return Dataset(
        X=X_crop,
        y=dataset.y,
        blocks=dataset.blocks,
        ch_names=dataset.ch_names,
        fs=dataset.fs,
    )

def preprocess_pipeline(dataset: Dataset) -> Dataset:
    """
    主前處理流程：
    - 重採樣到 cfg.fs
    - (必要時) 帶通 0.5–100 Hz, notch 50 Hz (你可自行加入)
    - 裁切到指定 trial 長度 (e.g. 1.5s)
    """
    ds = resample_to_fs(dataset, cfg.fs)
    # TODO: 如果需要加 Notch / 0.5–100Hz bandpass 可以在這裡加
    ds = crop_trials(ds, t_start=0.0, t_end=cfg.trial_len_sec)
    return ds
```

```python
@dataclass

    id: int
    start_idx: int
    end_idx: int

    id: int
    f_low: float
    f_high: float

    id: int
    name: str
    ch_idx: List[int]   # channel indices

class SFTSSpec:
    id: int
    ch_group_id: int
    freq_band_id: int
    time_window_id: int

def generate_time_windows() -> List[TimeWindow]:
    fs = cfg.fs
    total_len_samples = int(cfg.trial_len_sec * fs)
    windows: List[TimeWindow] = []
    wid = 0

    for L_sec in cfg.time_cfg.lengths:
        L = int(round(L_sec * fs))
        step = int(round(L * (1.0 - cfg.time_cfg.overlap)))  # e.g. 50% overlap = step = L/2
        if step == 0:
            step = 1
        start = 0
        while True:
            end = start + L
            if end > total_len_samples:
                break
            windows.append(TimeWindow(id=wid, start_idx=start, end_idx=end, length_sec=L_sec))
            wid += 1
            start += step

    return windows

def generate_freq_bands() -> List[FreqBand]:
    bands: List[FreqBand] = []
    bid = 0
    for bw in cfg.freq_cfg.bandwidths:
        f = cfg.freq_cfg.min_freq
        step = bw * (1.0 - cfg.freq_cfg.overlap)
        if step <= 0:
            step = bw
        while True:
            f_low = f
            f_high = f_low + bw
            if f_high > cfg.freq_cfg.max_freq + 1e-6:
                break
            bands.append(FreqBand(id=bid, f_low=f_low, f_high=f_high, bandwidth=bw))
```

```python
# features.py
from __future__ import annotations
from typing import Dict, List
import numpy as np

from config import cfg
from data_loader import Dataset
from preprocessing import bandpass_filter
from segmentation import (
    ChannelGroup, FreqBand, TimeWindow, SFTSSpec,
)
from divcsp import DivCSP, DivCSPParams

def precompute_freq_bands(
    dataset: Dataset,
    freq_bands: List[FreqBand],
) -> Dict[int, np.ndarray]:
    """
    對每個 freq_band 做一次 bandpass，回傳 dict:
        fb_id -> X_fband (n_trials, n_channels, n_samples)
    """
    X = dataset.X
    fs = dataset.fs
    X_fband = {}
    for fb in freq_bands:
        X_fband[fb.id] = bandpass_filter(X, fs, fb.f_low, fb.f_high)
    return X_fband

def get_sfts_data(
    X_fband: Dict[int, np.ndarray],
    sfts: SFTSSpec,
    channel_groups: List[ChannelGroup],
    time_windows: List[TimeWindow],
) -> np.ndarray:
    """
    根據 SFTS spec 從 precomputed frequency data 裡取出需要的片段。

    回傳: X_sfts, shape (n_trials, n_sel_channels, n_samples_window)
    """
    fb_X = X_fband[sfts.freq_band_id]  # (n_trials, n_channels, n_samples)

    cg = next(c for c in channel_groups if c.id == sfts.ch_group_id)
    tw = next(t for t in time_windows if t.id == sfts.time_window_id)

    X_c = fb_X[:, cg.ch_idx, :]                # (n_trials, n_sel_channels, n_samples)
    X_ct = X_c[..., tw.start_idx:tw.end_idx]   # (n_trials, n_sel_channels, n_samples_window)
    return X_ct

def extract_divcsp_features_for_sfts(
    dataset: Dataset,
    X_fband: Dict[int, np.ndarray],
    sfts: SFTSSpec,
    channel_groups: List[ChannelGroup],
    time_windows: List[TimeWindow],
    trial_idx: np.ndarray,
    y: np.ndarray,
    divcsp_params: DivCSPParams = None,  # 如果已有訓練好的參數，可直接用
) -> tuple[np.ndarray, DivCSPParams]:
    """
    在某個 SFTS 上，用 DivCSP 萃取指定 trial 的特徵。

    trial_idx: 要使用的 trial indices (e.g. training 或 validation indices)
    y: 全部 trial 的 label
    """
    X_sfts = get_sfts_data(X_fband, sfts, channel_groups, time_windows)
    X_sel = X_sfts[trial_idx]
    y_sel = y[trial_idx]

    divcsp = DivCSP()
    if divcsp_params is not None:
        divcsp.set_params(divcsp_params)
    else:
        divcsp.fit(X_sel, y_sel)
        divcsp_params = divcsp.get_params()

    feats = divcsp.transform(X_sel)
    return feats, divcsp_params
```

```python
# feature_selection.py
from __future__ import annotations
from dataclasses import dataclass
from typing import List, Dict, Tuple
import numpy as np
from sklearn.svm import LinearSVC
from sklearn.metrics import accuracy_score

from config import cfg
from data_loader import Dataset
from segmentation import SFTSSpec, ChannelGroup, TimeWindow, FreqBand
from features import precompute_freq_bands, get_sfts_data
from divcsp import DivCSP, DivCSPParams

@dataclass
class SFTSScore:
    sfts_id: int
    acc: float

@dataclass
class EnsembleMember:
    sfts_ids: List[int]                    # 使用的 SFTS (在 specs list 中的 index)
    svm: LinearSVC                         # 對應的 SVM 模型
    csp_params: Dict[int, DivCSPParams]    # sfts_id -> DivCSPParams

def lobo_blocks(dataset: Dataset) -> List[int]:
    """
    回傳所有 block id (用於 LOBO CV).
    """
    return sorted(np.unique(dataset.blocks).tolist())

def evaluate_single_sfts_lobo(
    dataset: Dataset,
    sfts_spec: SFTSSpec,
    X_fband: Dict[int, np.ndarray],
    channel_groups: List[ChannelGroup],
    time_windows: List[TimeWindow],
) -> float:
    """
    對單一 SFTS，用 LOBO 計算平均準確率，作為該 SFTS 的 score。
    這裡給 skeleton，你可以依照需求調整 (例如簡化計算，只 fit 一次 CSP 等)。
    """
    blocks = lobo_blocks(dataset)
    y = dataset.y
    acc_list = []

    for test_block in blocks:
        mask_test = dataset.blocks == test_block
        mask_train = ~mask_test

        trial_train = np.where(mask_train)[0]
        trial_test = np.where(mask_test)[0]

        # 1. 在 training trials 上 fit CSP
        X_sfts = get_sfts_data(X_fband, sfts_spec, channel_groups, time_windows)
        divcsp = DivCSP()
        divcsp.fit(X_sfts[trial_train], y[trial_train])

        f_train = divcsp.transform(X_sfts[trial_train])
        f_test = divcsp.transform(X_sfts[trial_test])

        clf = LinearSVC()
        clf.fit(f_train, y[trial_train])
        y_pred = clf.predict(f_test)
        acc_list.append(accuracy_score(y[trial_test], y_pred))

    return float(np.mean(acc_list)) if acc_list else 0.0

def rank_all_sfts(
    dataset: Dataset,
    sfts_specs: List[SFTSSpec],
    channel_groups: List[ChannelGroup],
    freq_bands: List[FreqBand],
    time_windows: List[TimeWindow],
) -> List[SFTSScore]:
    """
    評估每一個 SFTS 的 LOBO 準確率，並排序。
    """
    X_fband = precompute_freq_bands(dataset, freq_bands)
    scores: List[SFTSScore] = []

    for spec in sfts_specs:
        acc = evaluate_single_sfts_lobo(
            dataset=dataset,
            sfts_spec=spec,
            X_fband=X_fband,
            channel_groups=channel_groups,
            time_windows=time_windows,
        )
        scores.append(SFTSScore(sfts_id=spec.id, acc=acc))

    scores_sorted = sorted(scores, key=lambda s: s.acc, reverse=True)
    return scores_sorted

def build_ensemble(
    dataset: Dataset,
    sfts_specs: List[SFTSSpec],
    channel_groups: List[ChannelGroup],
    freq_bands: List[FreqBand],
    time_windows: List[TimeWindow],
    scores_sorted: List[SFTSScore],
) -> List[EnsembleMember]:
    """
    對應論文 Algorithm 2 的 merged SFTS 評估 & 取前 K 個 ensemble 成員。
    """
    X_fband = precompute_freq_bands(dataset, freq_bands)
    blocks = lobo_blocks(dataset)
    y = dataset.y

    AccMerged: List[float] = []
    members: List[EnsembleMember] = []

    D = cfg.fs_cfg.D
    K = cfg.fs_cfg.K

    # merged SFTS: j = D, 2D, 3D, ...
    for j in range(D, len(scores_sorted) + 1, D):
        top_j = scores_sorted[:j]
        sfts_ids = [s.sfts_id for s in top_j]

        # 這裡我們先簡化：對所有 trial 一次性萃取 features (每個 SFTS 都有一組 CSP)
        # 之後做 LOBO SVM。
        csp_params: Dict[int, DivCSPParams] = {}
        feats_all: Dict[int, np.ndarray] = {}  # sfts_id -> (n_trials, d)

        for s_id in sfts_ids:
            spec = next(s for s in sfts_specs if s.id == s_id)
            X_sfts = get_sfts_data(X_fband, spec, channel_groups, time_windows)
            divcsp = DivCSP()
            divcsp.fit(X_sfts, y)
            params = divcsp.get_params()
            csp_params[s_id] = params
            feats_all[s_id] = divcsp.transform(X_sfts)

        # LOBO 在 concatenated features 上
        acc_list = []
        for test_block in blocks:
            mask_test = dataset.blocks == test_block
            mask_train = ~mask_test
            idx_train = np.where(mask_train)[0]
            idx_test = np.where(mask_test)[0]

            feats_train_list = []
            feats_test_list = []
            for s_id in sfts_ids:
                F = feats_all[s_id]
                feats_train_list.append(F[idx_train])
                feats_test_list.append(F[idx_test])
            F_train = np.concatenate(feats_train_list, axis=1)
            F_test = np.concatenate(feats_test_list, axis=1)

            clf = LinearSVC()
            clf.fit(F_train, y[idx_train])
            y_pred = clf.predict(F_test)
            acc_list.append(accuracy_score(y[idx_test], y_pred))

        mean_acc = float(np.mean(acc_list)) if acc_list else 0.0
        AccMerged.append(mean_acc)

        # 這裡先存一個「在所有 trial 上重訓」的 SVM（方便最後 predict 時使用）
        # 也可以選擇只存当前 LOBO 最佳模型。
        feats_all_concat = np.concatenate(
            [feats_all[s_id] for s_id in sfts_ids],
            axis=1,
        )
        clf_final = LinearSVC()
        clf_final.fit(feats_all_concat, y)

        members.append(EnsembleMember(
            sfts_ids=sfts_ids,
            svm=clf_final,
            csp_params=csp_params,
        ))

    # 依 AccMerged 排序，取前 K 個 ensemble 成員
    idx_sorted = np.argsort(AccMerged)[::-1]
    top_members = [members[i] for i in idx_sorted[:K]]
    return top_members
```

## 8. `model.py`

```python
# model.py
from __future__ import annotations
from dataclasses import dataclass
from typing import List, Dict
import numpy as np

from data_loader import Dataset
from config import cfg
from segmentation import (
    generate_time_windows,
    generate_freq_bands,
    generate_channel_groups,
    generate_sfts_specs,
    SFTSSpec, ChannelGroup, TimeWindow, FreqBand,
)
from preprocessing import preprocess_pipeline, bandpass_filter
from feature_selection import rank_all_sfts, build_ensemble, EnsembleMember
from divcsp import DivCSP

@dataclass
class FGSFTMIModelState:
    channel_groups: List[ChannelGroup]
    freq_bands: List[FreqBand]
    time_windows: List[TimeWindow]
    sfts_specs: List[SFTSSpec]
    ensemble: List[EnsembleMember]

class FGSFTMIModel:
    def __init__(self):
        self.state: FGSFTMIModelState | None = None

    def fit(self, dataset: Dataset) -> "FGSFTMIModel":
        """
        Main training pipeline:
        - preprocess
        - generate S/F/T segments
        - wrapper-based feature selection -> ensemble
        """
        ds_proc = preprocess_pipeline(dataset)

        time_windows = generate_time_windows()
        freq_bands = generate_freq_bands()
        channel_groups = generate_channel_groups(ds_proc.ch_names)
        sfts_specs = generate_sfts_specs(channel_groups, freq_bands, time_windows)

        # 單一 SFTS ranking
        scores_sorted = rank_all_sfts(
            dataset=ds_proc,
            sfts_specs=sfts_specs,
            channel_groups=channel_groups,
            freq_bands=freq_bands,
            time_windows=time_windows,
        )

        # merged SFTS + ensemble building
        ensemble_members = build_ensemble(
            dataset=ds_proc,
            sfts_specs=sfts_specs,
            channel_groups=channel_groups,
            freq_bands=freq_bands,
            time_windows=time_windows,
            scores_sorted=scores_sorted,
        )

        self.state = FGSFTMIModelState(
            channel_groups=channel_groups,
            freq_bands=freq_bands,
            time_windows=time_windows,
            sfts_specs=sfts_specs,
            ensemble=ensemble_members,
        )
        return self

    def _precompute_freq_bands_for_predict(
        self,
        X: np.ndarray,
        fs: float,
        freq_bands: List[FreqBand],
    ) -> Dict[int, np.ndarray]:
        from preprocessing import bandpass_filter
        X_fband = {}
        for fb in freq_bands:
            X_fband[fb.id] = bandpass_filter(X, fs, fb.f_low, fb.f_high)
        return X_fband

    def predict_proba(self, dataset: Dataset) -> np.ndarray:
        """
        對新資料 (或測試資料) 做預測，輸出每一 trial 的 class 機率。
        """
        assert self.state is not None, "Call fit() first."
        ds_proc = preprocess_pipeline(dataset)

        st = self.state
        X = ds_proc.X
        fs = ds_proc.fs
        n_trials = X.shape[0]
        n_classes = 2

        # 頻帶預先濾波
        X_fband = self._precompute_freq_bands_for_predict(
            X, fs, st.freq_bands
        )

        # 對 ensemble 的每個成員計算機率
        prob_ensemble = np.zeros((len(st.ensemble), n_trials, n_classes), dtype=np.float32)

        from features import get_sfts_data
        from divcsp import DivCSP

        for m_idx, member in enumerate(st.ensemble):
            # 1. 對這個 member 需要的 SFTS 萃取 feature
            feats_list = []
            for s_id in member.sfts_ids:
                spec = next(s for s in st.sfts_specs if s.id == s_id)
                X_sfts = get_sfts_data(
                    X_fband,
                    spec,
                    st.channel_groups,
                    st.time_windows,
                )
                divcsp = DivCSP()
                divcsp.set_params(member.csp_params[s_id])
                feats_list.append(divcsp.transform(X_sfts))
            F_all = np.concatenate(feats_list, axis=1)

            # 2. 用對應 SVM 取 decision function -> 轉成機率
            df = member.svm.decision_function(F_all)  # (n_trials,)
            # binary: 使用 sigmoid 做 soft 轉換
            # TODO: 這裡可以用 Platt scaling 等方式作更好的 probability calibration
            p1 = 1.0 / (1.0 + np.exp(-df))
            p0 = 1.0 - p1
            prob_ensemble[m_idx, :, 0] = p0
            prob_ensemble[m_idx, :, 1] = p1

        prob_mean = prob_ensemble.mean(axis=0)
        return prob_mean

    def predict(self, dataset: Dataset) -> np.ndarray:
        prob = self.predict_proba(dataset)
        return np.argmax(prob, axis=1)
```

## 9. `metrics.py`

```python
# metrics.py
import numpy as np

def accuracy(y_true: np.ndarray, y_pred: np.ndarray) -> float:
    return float((y_true == y_pred).mean()) if len(y_true) > 0 else 0.0

def compute_itr(P: float, N: int, T: float) -> float:
    """
    計算 ITR (bits/min):
        ITR = ( log2(N) + P*log2(P) + (1-P)*log2((1-P)/(N-1)) ) * (60 / T)

    P: accuracy (0~1)
    N: number of classes
    T: 每次 trial 所需時間 (秒)
    """
    if P <= 0 or P >= 1 or N <= 1:
        return 0.0
    import math
    return (math.log2(N) +
            P * math.log2(P) +
            (1 - P) * math.log2((1 - P) / (N - 1))) * (60.0 / T)
```
